/**1.Total Customers**/
SELECT COUNT(*) AS total_customers
FROM telecom_customers;

/**2. Churn Count & Churn Rate**/
SELECT 
    "Churn", 
    COUNT(*) AS customer_count,
    ROUND(100.0 * COUNT(*) / SUM(COUNT(*)) OVER(), 2) AS churn_percentage
FROM telecom_customers
GROUP BY "Churn"
ORDER BY customer_count DESC;

/**3. Average Monthly Charges by Churn**/
SELECT 
    "Churn", 
    ROUND(AVG("MonthlyCharges")::numeric, 2) AS avg_monthly_charges
FROM telecom_customers
GROUP BY "Churn";

/**4. Average Tenure by Churn**/
SELECT 
    "Churn", 
    ROUND(AVG("tenure")::numeric, 1) AS avg_tenure
FROM telecom_customers
GROUP BY "Churn";

/**5. Churn by Contract Type**/
SELECT 
    "Contract", 
    COUNT(*) AS total_customers,
    SUM(CASE WHEN "Churn" = 'Yes' THEN 1 ELSE 0 END) AS churned_customers,
    ROUND(100.0 * SUM(CASE WHEN "Churn" = 'Yes' THEN 1 ELSE 0 END)::numeric / COUNT(*), 2) AS churn_rate
FROM telecom_customers
GROUP BY "Contract"
ORDER BY churn_rate DESC;

/**6. Payment Method Impact on Churn**/
SELECT 
    "PaymentMethod",
    COUNT(*) AS total_customers,
    SUM(CASE WHEN "Churn" = 'Yes' THEN 1 ELSE 0 END) AS churned_customers,
    ROUND(100.0 * SUM(CASE WHEN "Churn" = 'Yes' THEN 1 ELSE 0 END)::numeric / COUNT(*), 2) AS churn_rate
FROM telecom_customers
GROUP BY "PaymentMethod"
ORDER BY churn_rate DESC;

/**7. Top 5 Customers by Total Charges**/
SELECT 
    "customerID", 
    "TotalCharges"
FROM telecom_customers
WHERE "TotalCharges" IS NOT NULL
  AND trim("TotalCharges") <> ''  
ORDER BY "TotalCharges"::numeric DESC
LIMIT 5;

/**8. Churn by Internet Service**/
SELECT 
    "InternetService",
    COUNT(*) AS total_customers,
    SUM(CASE WHEN "Churn" = 'Yes' THEN 1 ELSE 0 END) AS churned_customers,
    ROUND(100.0 * SUM(CASE WHEN "Churn" = 'Yes' THEN 1 ELSE 0 END)::numeric / COUNT(*), 2) AS churn_rate
FROM telecom_customers
GROUP BY "InternetService"
ORDER BY churn_rate DESC;

/**9. Churn by Senior Citizen**/
SELECT 
    "SeniorCitizen",
    COUNT(*) AS total_customers,
    SUM(CASE WHEN "Churn" = 'Yes' THEN 1 ELSE 0 END) AS churned_customers,
    ROUND(100.0 * SUM(CASE WHEN "Churn" = 'Yes' THEN 1 ELSE 0 END)::numeric / COUNT(*), 2) AS churn_rate
FROM telecom_customers
GROUP BY "SeniorCitizen";

/**10. Revenue by Contract Type**/
SELECT 
    "Contract",
    ROUND(SUM("MonthlyCharges")::numeric, 2) AS total_revenue
FROM telecom_customers
GROUP BY "Contract"
ORDER BY total_revenue DESC;
